Quick Start
Make your first API call with your API key pre-filled.
curl https://docuextract-azure.vercel.app/v1/extract \
  -H "Authorization: Bearer YOUR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"document":"https://example.com/invoice.pdf","type":"invoice"}'
import DocuExtract from 'docuextract';

const dex = new DocuExtract('YOUR_API_KEY');
const result = await dex.extract('https://example.com/invoice.pdf', {
  type: 'invoice'
});
console.log(result.data);
from docuextract import DocuExtract

dex = DocuExtract('YOUR_API_KEY')
result = dex.extract('https://example.com/invoice.pdf', type='invoice')
print(result.data)
